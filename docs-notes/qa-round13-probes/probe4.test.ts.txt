// TEMPORARY QA probe 4 — ReDoS, CLI escape, misc. Delete after the review.
import { mkdtemp, rm } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { afterEach, beforeEach, describe, expect, it } from "vitest";
import { matchesPattern } from "./pattern-match.js";
import { evaluateGovernancePolicy } from "./policy-engine.js";
import { loadPolicy, savePolicy } from "./policy-store.js";
import { validateRulePattern } from "./rule-validation.js";

let dir: string;
let workspace: string;
beforeEach(async () => {
  dir = await mkdtemp(join(tmpdir(), "gov-probe4-"));
  process.env.OPENCLAW_GOVERNANCE_DIR = dir;
  workspace = await mkdtemp(join(tmpdir(), "gov-probe4-ws-"));
});
afterEach(async () => {
  delete process.env.OPENCLAW_GOVERNANCE_DIR;
  await rm(dir, { recursive: true, force: true });
  await rm(workspace, { recursive: true, force: true });
});

const ctx = () => ({ agentId: "agent-a", sessionKey: "agent:agent-a:main", cwd: workspace });
function verdict(d: Awaited<ReturnType<typeof evaluateGovernancePolicy>>): string {
  if (!d) {
    return "allow";
  }
  return "block" in d ? "block" : "ask";
}

describe("PROBE 12 — ReDoS patterns the validator accepts", () => {
  const patterns = [
    "^(a|a)+$",
    "^(a|ab)+$",
    "^([a-z]|[a-z])+$",
    "^(.*a){20}$",
    "^(x|x|x)+y$",
  ];
  for (const pattern of patterns) {
    it(`accepts and then hangs on ${pattern}`, () => {
      const validation = validateRulePattern(pattern);
      if (!validation.ok) {
        console.log(`REDOS ${pattern.padEnd(20)} -> REJECTED (${validation.error})`);
        return;
      }
      const victim = `${"a".repeat(30)}!`;
      const started = Date.now();
      matchesPattern(pattern, victim);
      const elapsed = Date.now() - started;
      console.log(`REDOS ${pattern.padEnd(20)} -> ACCEPTED, match took ${elapsed}ms`);
      expect(elapsed).toBeLessThan(250);
    });
  }
});

describe("PROBE 13 — reaching the governance CLI from a shell", () => {
  const commands = [
    "node openclaw.mjs governance policy set-mode off",
    "npx openclaw governance policy set-mode off",
    "openclaw governance policy set-mode off",
    "pnpm openclaw governance policy set-mode off",
  ];
  for (const command of commands) {
    it(`[${command}]`, async () => {
      const doc = await loadPolicy();
      await savePolicy({
        ...doc,
        mode: "enforce",
        ask: "off",
        rules: [
          ...doc.rules,
          {
            id: "probe-broad",
            resourceKind: "command",
            effect: "allow",
            tier: "admin",
            // The kind of rule an operator writes so their agent can run the project.
            pattern: "^(node|npm|npx|pnpm|openclaw) .*$",
            createdAt: new Date().toISOString(),
            createdBy: "probe",
          } as never,
        ],
      });
      const d = await evaluateGovernancePolicy({ toolName: "exec", params: { command } }, ctx());
      console.log(`CLI-ESCAPE ${command.padEnd(48)} -> ${verdict(d)}`);
      expect(verdict(d)).toBe("block");
    });
  }
});

describe("PROBE 14 — rule count and pattern-cache pressure", () => {
  it("keeps evaluation fast with a full ruleset", async () => {
    const doc = await loadPolicy();
    const many = Array.from({ length: 500 }, (_, index) => ({
      id: `probe-${index}`,
      resourceKind: "command" as const,
      effect: "allow" as const,
      tier: "admin" as const,
      pattern: `^never-matches-${index}-[a-z]{1,20}$`,
      createdAt: new Date().toISOString(),
      createdBy: "probe",
    }));
    await savePolicy({ ...doc, mode: "enforce", ask: "off", rules: [...doc.rules, ...many] });
    const started = Date.now();
    for (let i = 0; i < 20; i += 1) {
      await evaluateGovernancePolicy({ toolName: "exec", params: { command: `ls ${i}` } }, ctx());
    }
    const elapsed = Date.now() - started;
    console.log(`RULESET 500 rules, 20 calls -> ${elapsed}ms`);
    expect(elapsed).toBeLessThan(5000);
  });
});

describe("PROBE 15 — resource shapes the extractor may miss", () => {
  const cases: Array<[string, Record<string, unknown>]> = [
    ["exec array command", { command: ["sudo", "-i"] }],
    ["exec nested", { command: { toString: "sudo" } }],
    ["exec empty", { command: "" }],
    ["web_fetch no url", { uri: "http://169.254.169.254/" }],
    ["read numeric path", { path: 123 }],
  ];
  for (const [label, params] of cases) {
    it(`[${label}]`, async () => {
      const doc = await loadPolicy();
      await savePolicy({ ...doc, mode: "enforce", ask: "off" });
      const toolName = label.startsWith("exec")
        ? "exec"
        : label.startsWith("web_fetch")
          ? "web_fetch"
          : "read";
      const d = await evaluateGovernancePolicy({ toolName, params }, ctx());
      console.log(`SHAPE ${label.padEnd(22)} -> ${verdict(d)}`);
      expect(verdict(d)).toBe("block");
    });
  }
});
