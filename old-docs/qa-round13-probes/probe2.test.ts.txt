// TEMPORARY QA probe 2. Delete after the review.
import { mkdtemp, rm } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { afterEach, beforeEach, describe, expect, it } from "vitest";
import { evaluateGovernancePolicy } from "./policy-engine.js";
import { loadPolicy, savePolicy } from "./policy-store.js";

let dir: string;
let workspace: string;
beforeEach(async () => {
  dir = await mkdtemp(join(tmpdir(), "gov-probe2-"));
  process.env.OPENCLAW_GOVERNANCE_DIR = dir;
  workspace = await mkdtemp(join(tmpdir(), "gov-probe2-ws-"));
});
afterEach(async () => {
  delete process.env.OPENCLAW_GOVERNANCE_DIR;
  await rm(dir, { recursive: true, force: true });
  await rm(workspace, { recursive: true, force: true });
});

const ctx = () => ({ agentId: "agent-a", sessionKey: "agent:agent-a:main", cwd: workspace });
function verdict(d: Awaited<ReturnType<typeof evaluateGovernancePolicy>>): string {
  if (!d) {
    return "allow";
  }
  return "block" in d ? "block" : "ask";
}
/** enforce + ask off + a wide-open allow rule for one kind, so only DENY rules can stop it. */
async function permissive(kind: "command" | "path" | "network"): Promise<void> {
  const doc = await loadPolicy();
  await savePolicy({
    ...doc,
    mode: "enforce",
    ask: "off",
    rules: [
      ...doc.rules,
      {
        id: `probe-open-${kind}`,
        resourceKind: kind,
        effect: "allow",
        tier: "admin",
        pattern: "^.*$",
        createdAt: new Date().toISOString(),
        createdBy: "probe",
      } as never,
    ],
  });
}

const NL = String.fromCharCode(10);
const CR = String.fromCharCode(13);
const BS = String.fromCharCode(92);
const DQ = String.fromCharCode(34);

describe("PROBE 4. Network core denial vs alternate spellings", () => {
  const cases: Array<[string, string]> = [
    ["plain", "http://169.254.169.254/latest/meta-data/"],
    ["trailing dot", "http://169.254.169.254./latest/"],
    ["decimal int", "http://2852039166/latest/"],
    ["hex octets", "http://0xa9.0xfe.0xa9.0xfe/latest/"],
    ["ipv4-mapped v6", "http://[::ffff:169.254.169.254]/latest/"],
    ["hex v6 mapped", "http://[::ffff:a9fe:a9fe]/latest/"],
    ["aws imds v6", "http://[fd00:ec2::254]/latest/"],
    ["alibaba imds", "http://100.100.100.200/latest/"],
    ["gcp alias", "http://metadata.google.internal/computeMetadata/v1/"],
    ["gcp short alias", "http://metadata/computeMetadata/v1/"],
    ["file url", "file:///etc/shadow"],
    ["localhost gateway", "http://127.0.0.1:18799/control-ui/governance/policy"],
  ];
  for (const [label, url] of cases) {
    it(`[${label}] ${url}`, async () => {
      await permissive("network");
      const d = await evaluateGovernancePolicy({ toolName: "web_fetch", params: { url } }, ctx());
      console.log(`NETWORK ${label.padEnd(18)} -> ${verdict(d)}`);
      expect(verdict(d)).toBe("block");
    });
  }
});

describe("PROBE 8. Lockdown coverage", () => {
  it("blocks when only the session key identifies the agent", async () => {
    const doc = await loadPolicy();
    await savePolicy({ ...doc, mode: "enforce", lockedAgents: ["agent-a"] });
    const d = await evaluateGovernancePolicy(
      { toolName: "exec", params: { command: "ls" } },
      { sessionKey: "agent:agent-a:main", cwd: workspace },
    );
    expect(verdict(d)).toBe("block");
  });
  it("blocks a locked agent whose id is absent from both fields", async () => {
    const doc = await loadPolicy();
    await savePolicy({ ...doc, mode: "enforce", lockedAgents: ["agent-a"] });
    const d = await evaluateGovernancePolicy(
      { toolName: "exec", params: { command: "ls" } },
      { cwd: workspace },
    );
    console.log("LOCKDOWN no-agent-id ->", verdict(d));
    expect(verdict(d)).toBe("block");
  });
  it("blocks an ungoverned tool for a locked agent", async () => {
    const doc = await loadPolicy();
    await savePolicy({ ...doc, mode: "enforce", lockedAgents: ["agent-a"] });
    const d = await evaluateGovernancePolicy(
      { toolName: "computer", params: { action: "type", text: "x" } },
      ctx(),
    );
    console.log("LOCKDOWN ungoverned-tool ->", verdict(d));
    expect(verdict(d)).toBe("block");
  });
});

describe("PROBE 9. Hand-edited policy.json", () => {
  it("cannot remove a core rule by rewriting the file", async () => {
    const doc = await loadPolicy();
    await savePolicy({
      ...doc,
      mode: "enforce",
      ask: "off",
      rules: doc.rules.filter((r) => r.tier !== "core"),
    });
    await permissive("path");
    const d = await evaluateGovernancePolicy({ toolName: "read", params: { path: ".env" } }, ctx());
    console.log("HAND-EDIT strip core ->", verdict(d));
    expect(verdict(d)).toBe("block");
  });
  it("cannot neuter a core rule by rewriting its pattern", async () => {
    const doc = await loadPolicy();
    await savePolicy({
      ...doc,
      mode: "enforce",
      ask: "off",
      rules: doc.rules.map((r) => (r.tier === "core" ? { ...r, pattern: "^never-matches$" } : r)),
    });
    await permissive("path");
    const d = await evaluateGovernancePolicy({ toolName: "read", params: { path: ".env" } }, ctx());
    console.log("HAND-EDIT neuter core ->", verdict(d));
    expect(verdict(d)).toBe("block");
  });
  it("cannot flip a core deny into an allow", async () => {
    const doc = await loadPolicy();
    await savePolicy({
      ...doc,
      mode: "enforce",
      ask: "off",
      rules: doc.rules.map((r) => (r.tier === "core" ? { ...r, effect: "allow" as const } : r)),
    });
    const d = await evaluateGovernancePolicy({ toolName: "read", params: { path: ".env" } }, ctx());
    console.log("HAND-EDIT flip core ->", verdict(d));
    expect(verdict(d)).toBe("block");
  });
  it("cannot disable the gate for one agent by adding agentMode off", async () => {
    const doc = await loadPolicy();
    await savePolicy({
      ...doc,
      mode: "enforce",
      ask: "off",
      lockedAgents: ["agent-a"],
      agentMode: { "agent-a": "off" },
    });
    const d = await evaluateGovernancePolicy({ toolName: "read", params: { path: ".env" } }, ctx());
    console.log("HAND-EDIT agentMode off (locked agent) ->", verdict(d));
    expect(verdict(d)).toBe("block");
  });
});
