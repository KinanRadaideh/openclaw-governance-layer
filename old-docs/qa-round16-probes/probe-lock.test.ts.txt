// PROBE. Round sixteen, T11. Not part of the suite; kept as the artefact that
// produced findings 104-106. Run by copying to src/governance/probe.test.ts.
//
// T11 was recorded as one observation: "a lock reclaimable from a slow writer.
// Stale locks are reclaimed after 15 s with no heartbeat. Fine while critical
// sections are milliseconds; not guaranteed on a loaded host with a large
// ledger."
//
// The probe asks whether that is the whole of it. It is not. Being reaped is
// the *start* of the failure, not the end, because nothing tells the reaped
// holder it has been reaped.
//
// The probes backdate the lock file's mtime rather than sleeping 15 seconds.
// That is the same condition the reaper tests for and makes the race
// deterministic instead of load-dependent.
import { mkdtemp, readFile, rm, stat, utimes, writeFile } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { afterEach, beforeEach, expect, it } from "vitest";
import { STALE_LOCK_MS_FOR_TESTS, withFileLock } from "../governance/file-lock.js";

let dir: string;
let target: string;
let lockPath: string;

beforeEach(async () => {
  dir = await mkdtemp(join(tmpdir(), "probe-lock-"));
  target = join(dir, "ledger.jsonl");
  lockPath = `${target}.lock`;
  await writeFile(target, "");
});

afterEach(async () => {
  await rm(dir, { recursive: true, force: true });
});

/** Makes the live lock look older than the staleness threshold. */
async function backdateLock(): Promise<void> {
  const past = new Date(Date.now() - STALE_LOCK_MS_FOR_TESTS - 5_000);
  await utimes(lockPath, past, past);
}

// FINDING 104. A slow holder is reaped, and is never told.
//
// Expected before the fix: two callers run their critical sections at the same
// time, which is the one thing the lock exists to prevent.
it("PROBE 104: a reaped holder keeps running as though it still holds the lock", async () => {
  const inside: string[] = [];
  let releaseA: () => void = () => {};
  const aInside = new Promise<void>((resolve) => {
    releaseA = resolve;
  });

  const a = withFileLock(target, async () => {
    inside.push("a-enter");
    await backdateLock(); // A is now indistinguishable from a crashed process.
    await aInside;
    inside.push("a-exit");
  });

  // Let A enter and backdate.
  await new Promise((r) => setTimeout(r, 50));

  const b = withFileLock(target, async () => {
    inside.push("b-enter");
    // If this runs while A is still inside, mutual exclusion is gone.
    expect(inside).not.toContain("a-exit");
    inside.push("b-exit");
  });

  await b;
  releaseA();
  await a;

  // The damning sequence: a-enter, b-enter, b-exit, a-exit.
  expect(inside).toEqual(["a-enter", "b-enter", "b-exit", "a-exit"]);
});

// FINDING 105. The reaped holder then deletes somebody else's lock.
//
// This is the worse half. `releaseLock` does `rm(lockPath, { force: true })`
// with no check that the file it is removing is the one it created. A holder
// that was reaped therefore unlocks whoever reaped it, on its way out.
it("PROBE 105: a reaped holder's release deletes the new holder's lock", async () => {
  let releaseA: () => void = () => {};
  const aInside = new Promise<void>((resolve) => {
    releaseA = resolve;
  });

  const a = withFileLock(target, async () => {
    await backdateLock();
    await aInside;
  });
  await new Promise((r) => setTimeout(r, 50));

  // B reaps A and takes the lock. B stays inside.
  let releaseB: () => void = () => {};
  const bInside = new Promise<void>((resolve) => {
    releaseB = resolve;
  });
  let bHeld = false;
  const b = withFileLock(target, async () => {
    bHeld = true;
    await bInside;
  });
  await new Promise((r) => setTimeout(r, 50));
  expect(bHeld).toBe(true);

  // A now exits. Its `finally` removes the lock file, which is B's.
  releaseA();
  await a;

  // B is still inside its critical section, and its lock is gone.
  const lockGone = await stat(lockPath).then(
    () => false,
    () => true,
  );
  expect(lockGone).toBe(true); // fails after the fix: A must not delete B's lock

  releaseB();
  await b;
});

// FINDING 106, two waiters can both reap, and the second deletes the first's
// freshly acquired lock.
//
// `reapStaleLock` is stat-then-remove with no compare: the file it removes need
// not be the file it judged stale. Two waiters that both see the stale lock
// both call `rm`, and the second one lands after the first has already won the
// lock and created a fresh file under the same name.
it("PROBE 106: reaping is stat-then-delete with no identity check", async () => {
  // A stale lock left by a crashed process.
  await writeFile(lockPath, "");
  await backdateLock();

  const order: string[] = [];
  const runners = [1, 2, 3].map((n) =>
    withFileLock(target, async () => {
      order.push(`enter-${n}`);
      await new Promise((r) => setTimeout(r, 30));
      order.push(`exit-${n}`);
    }),
  );
  await Promise.all(runners);

  // Every enter must be immediately followed by its own exit. Interleaving
  // means two ran at once.
  for (let index = 0; index < order.length; index += 2) {
    const enter = order[index]?.split("-")[1];
    const exit = order[index + 1]?.split("-")[1];
    expect(exit).toBe(enter);
  }
});

// A control, to be sure the probe is measuring the lock and not the harness.
it("PROBE control: without backdating, the lock serializes correctly", async () => {
  const order: string[] = [];
  await Promise.all(
    [1, 2, 3].map((n) =>
      withFileLock(target, async () => {
        order.push(`enter-${n}`);
        await new Promise((r) => setTimeout(r, 20));
        order.push(`exit-${n}`);
      }),
    ),
  );
  for (let index = 0; index < order.length; index += 2) {
    expect(order[index + 1]?.split("-")[1]).toBe(order[index]?.split("-")[1]);
  }
});
